8.Prime Number

bool isPrime(int number) {
  if (number <= 1) return false; // 0 and 1 are not prime
  if (number == 2) return true;  // 2 is prime

  for (int i = 2; i <= number ~/ 2; i++) {
    if (number % i == 0) {
      return false;
    }
  }

  return true;
}

void main(){
  print(isPrime(2));
}

12.Frequency of words in a sentence using a map

void main() {
  String sentence = "hello world hello dart world";

  // Split sentence into words
  List<String> words = sentence.split(" ");

  // Create a map to hold word frequencies
  Map<String, int> wordCount = {};

  for (String word in words) {
    if (wordCount.containsKey(word)) {
      wordCount[word] = wordCount[word]! + 1;
    } else {
      wordCount[word] = 1;
    }
  }

  // Print word frequencies
  wordCount.forEach((word, count) {
    print('$word: $count');
  });
}

16.Polymorphism:


// Base class
class Animal {
  void makeSound() {
    print("Some generic animal sound");
  }
}

// Derived class 1
class Dog extends Animal {
  @override
  void makeSound() {
    print("Dog barks");
  }
}

// Derived class 2
class Cat extends Animal {
  @override
  void makeSound() {
    print("Cat meows");
  }
}

void main() {
  Animal myDog = Dog();  // Polymorphism: Animal reference, Dog object
  Animal myCat = Cat();  // Polymorphism: Animal reference, Cat object

  myDog.makeSound();  // Output: Dog barks
  myCat.makeSound();  // Output: Cat meows
}

18.Binary Search

int binarySearch(List<int> arr, int target, int low, int high) {
  if (low > high) return -1; // Element not found

  int mid = (low + high) ~/ 2;

  if (arr[mid] == target) return mid;
  else if (target < arr[mid]) {
    return binarySearch(arr, target, low, mid - 1);
  } else {
    return binarySearch(arr, target, mid + 1, high);
  }
}

void main() {
  List<int> nums = [1, 3, 5, 7, 9, 11];
  int target = 7;
  int result = binarySearch(nums, target, 0, nums.length - 1);

  if (result != -1) {
    print("Element found at index: $result");
  } else {
    print("Element not found");
  }
}
