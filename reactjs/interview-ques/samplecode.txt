app.js

import react from 'react';
import{useState} from 'react';
import './home.css'
export default function App(){
const [count,setCount] = useState(0);
return(
  <>
  <div className="Counter-Container">
  <h1>{count}</h1>
  <div className="counterBtn">
  <button className="buttonInc" onClick={()=>setCount(prev =>prev+1)}>Increment</button>
  <button className="buttonDec" disabled={count <= 0}  onClick={()=>setCount(prev=>prev-1)}>Decrement</button>
   </div>
  </div>
  </>
)
}
